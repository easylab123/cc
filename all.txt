import socket, threading
s = socket.socket()
s.bind(("127.0.0.1", 8000))
s.listen()
clients = []
def handle(c):
while True:
try:
m = c.recv(1024)
if not m: break
for x in clients:
if x != c: x.sendall(m)
except: break
if c in clients: clients.remove(c)
c.close()
while True:
c, _ = s.accept()
clients.append(c)
threading.Thread(target=handle, args=(c,), daemon=True).start()

Task 2: Distributed Mutual Exclusion:
A. Ricart-Agrawala Algorithm:
import threading, queue, time
N = 3
q = [queue.Queue() for _ in range(N)]
def process(p):
clock, ts = 0, 0
requesting, in_cs = False, False
pending, deferred = set(), []
lock = threading.Lock()
event = threading.Event()
def listen():
nonlocal clock
while True:
m = q[p].get()
with lock:
if m[0] == "REQ":
_, t, j = m
clock = max(clock, t) + 1
if in_cs or (requesting and (ts, p) < (t, j)):
deferred.append(j)
else:
clock += 1
q[j].put(("REP", p, clock))
else:
clock = max(clock, m[2]) + 1
pending.discard(m[1])
if not pending:
event.set()
threading.Thread(target=listen, daemon=True).start()
for _ in range(2):
with lock:
clock += 1
ts, requesting = clock, True
pending = set(range(N)) - {p}
event.clear()
for j in pending:
q[j].put(("REQ", ts, p))
event.wait()
with lock:
in_cs = True
print(f"P{p} enters CS")
time.sleep(0.1)
print(f"P{p} leaves CS")
with lock:
in_cs, requesting = False, False
for j in deferred:
clock += 1
q[j].put(("REP", p, clock))
deferred.clear()
threads = [threading.Thread(target=process, args=(p,)) for p in range(N)]
for t in threads:
t.start()
for t in threads:
t.join()
print("Ricart-Agrawala complete")

B. Token Ring Algorithm:
import threading, queue, time, random
N = 5
q = [queue.Queue() for _ in range(N)]
def process(pid):
for _ in range(3):
q[pid].get()
print(f"P{pid} received token")
if random.random() < 0.6:
print(f"P{pid} enters CS")
time.sleep(0.2)
print(f"P{pid} leaves CS")
q[(pid + 1) % N].put("TOKEN")
threads = [threading.Thread(target=process, args=(i,)) for i in range(N)]
for t in threads:
t.start()
q[0].put("TOKEN")
for t in threads:
t.join()
print("Token ring complete")

C. Centralized Coordinator Algorithm:
import threading, queue, time
N = 3
coord = queue.Queue()
q = [queue.Queue() for _ in range(N)]
def coordinator():
busy, waiting = False, []
while True:
msg, p = coord.get()
if msg == "REQ":
if not busy:
busy = True
q[p].put("GRANT")
else:
waiting.append(p)
elif waiting:
q[waiting.pop(0)].put("GRANT")
else:
busy = False
def process(p):
for _ in range(2):
coord.put(("REQ", p))
q[p].get()
print(f"P{p} enters CS")
time.sleep(0.1)
print(f"P{p} leaves CS")
coord.put(("REL", p))
threading.Thread(target=coordinator, daemon=True).start()
ts = [threading.Thread(target=process, args=(p,)) for p in range(N)]
for t in ts: t.start()
for t in ts: t.join()

class LamportClock:
def __init__(self, process_id):
self.process_id = process_id
self.clock = 0
def event(self):
self.clock += 1
print(f"Process {self.process_id} - Internal event "
f"with Timestamp {self.clock}")
def send_message(self, receiver):
self.clock += 1
print(f"Process {self.process_id} sends a message "
f"to Process {receiver.process_id} "
f"with Timestamp {self.clock}")
receiver.receive_message(self.process_id, self.clock)
def receive_message(self, sender_id, sender_timestamp):
self.clock = max(self.clock, sender_timestamp) + 1
print(f"Process {self.process_id} received a message "
f"from Process {sender_id} with updated "
f"Timestamp {self.clock}")

Task 2: Vector Clocks:
class VectorClock:
def __init__(self, process_id, num_processes):
self.process_id = process_id
self.num_processes = num_processes
self.vector = [0] * num_processes
def event(self):
self.vector[self.process_id] += 1
print(f"Process {self.process_id} - Internal event "
f"with Updated Vector {self.vector}")
def send_message(self, receiver):
self.vector[self.process_id] += 1
print(f"Process {self.process_id} sends a message "
f"to Process {receiver.process_id} "
f"with Updated Vector {self.vector}")
receiver.receive_message(self.process_id, self.vector.copy())
def receive_message(self, sender_id, sender_vector):
for i in range(self.num_processes):
self.vector[i] = max(self.vector[i], sender_vector[i])
self.vector[self.process_id] += 1
print(f"Process {self.process_id} received a message "
f"from Process {sender_id} with Updated "
f"Vector {self.vector}")
# Create Processes
process1 = VectorClock(0, 3)
process2 = VectorClock(1, 3)
process3 = VectorClock(2, 3)
# Simulate Events
process1.event()
process1.send_message(process2)
process2.event()
process2.send_message(process3)
process3.send_message(process1)

The client sends a filename to the server. The server checks whether the file
exists and returns the number of lines, number of words, and size of the file
to the client.

import socket
import os
server = socket.socket()
server.bind(("127.0.0.1", 5000))
server.listen(1)
print("Server is waiting for client...")
conn, addr = server.accept()
filename = conn.recv(1024).decode()
if os.path.isfile(filename):
with open(filename, "r") as f:
data = f.read()
lines = len(data.splitlines())
words = len(data.split())
size = os.path.getsize(filename)
result = f"Lines: {lines}\nWords: {words}\nSize: {size} bytes"

result = "File does not exist"

Berkeley's clock synchronization algorithm
Implement Berkeley's algorithm to synchronize the clocks of multiple
distributed systems by calculating their clock offsets and adjusting the
clocks using the coordinator's calculated average time.
server.py
import socket
server = socket.socket()
server.bind(("0.0.0.0", 5000))
server.listen(5)
coordinator_time = int(input("Enter coordinator clock: "))
n = int(input("Enter number of clients: "))
clocks = [coordinator_time]
connections = []
print("Waiting for clients...")
# Receive clock times
for i in range(n):
conn, addr = server.accept()
connections.append(conn)
time = int(conn.recv(1024).decode())
clocks.append(time)
# Calculate clock offsets
offsets = []
for t in clocks:
offsets.append(t - coordinator_time)
print("Clock times:", clocks)
print("Offsets:", offsets)
# Calculate average offset
avg = sum(offsets) / len(offsets)
print("Average offset:", avg)
# Adjust coordinator clock
coordinator_time = coordinator_time + avg
print("Synchronized coordinator clock:", coordinator_time)
# Send adjustments to clients
for i in range(n):
adjustment = avg - offsets[i + 1]
connections[i].send(str(adjustment).encode())
print("Adjustment sent to client", i + 1, ":", adjustment)
connections[i].close()
server.close()

Physical clock drift simulation
Simulate two clocks running on different systems. Assume both clocks start
at the same time, but Clock B runs faster by 10 milliseconds per second.
Write a Python program to simulate the drift over one minute and calculate
how far apart the two clocks become.
import time
clock_a = 0.0
clock_b = 0.0
for i in range(60):
time.sleep(1)
clock_a += 1.0
clock_b += 1.01
drift = clock_b - clock_a
print("Time:", i + 1, "seconds")
print("Clock A:", round(clock_a, 2))
print("Clock B:", round(clock_b, 2))
print("Drift:", round(drift * 1000), "milliseconds")
print()
print("Final drift:", round(drift * 1000), "milliseconds")
Round-robin task scheduler simulation
There are two worker nodes. Create multiple tasks with different processing
requirements. Assign tasks to each worker using the round-robin algorithm.
Simulate the execution of each task, find its completion time, and print the
number of tasks assigned to each worker.
# Round-Robin Task Scheduling with Two Workers
tasks = [4, 2, 6, 3, 5]
workers = [0, 0]
count = [0, 0]
print("Round-Robin Task Scheduling\n")
for i in range(len(tasks)):
# Assign tasks alternately
w = i % 2
# Start and completion times
start_time = workers[w]
completion_time = start_time + tasks[i]
# Update worker's available time
workers[w] = completion_time
# Count assigned tasks
count[w] += 1
# Display task details
print("Task:", i + 1)
print("Processing time:", tasks[i])
print("Assigned to Worker:", w + 1)
print("Start time:", start_time)
print("Completion time:", completion_time)
print()
# Display final results
print("Total tasks assigned to Worker 1:", count[0])
print("Total tasks assigned to Worker 2:", count[1])
print("Worker 1 finishes at:", workers[0])
print("Worker 2 finishes at:", workers[1])

#!/bin/bash
(sleep 3; echo "Process 1 executed after 3 seconds" >> output.log) &
(sleep 15; echo "Process 2 executed after 15 seconds" >> output.log) &
wait
2. Make the script executable
chmod +x processes.sh
3. Configure cron
Open the crontab editor:
crontab -e
Add this line, replacing /full/path/to/processes.sh with the script's actual absolute path:
* * * * * /bin/bash /full/path/to/processes.sh
Important: This runs the script every minute. It does not run just once. For a lab demonstration,
you can temporarily use this entry, verify the output, and remove the entry afterward.
